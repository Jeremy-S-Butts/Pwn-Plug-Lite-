#!/usr/bin/env python3
"""
Password Policy Tester (authorized use)

- DOES NOT attempt logins or authentication.
- Evaluates candidate passwords against a policy and produces a report.
- Optionally loads local Linux policy hints (pwquality.conf, login.defs).

Usage examples:
  python3 pw_policy_tester.py --passwords passwords.txt --users users.txt --report report.json
  python3 pw_policy_tester.py --passwords passwords.txt --username alice
  python3 pw_policy_tester.py --passwords passwords.txt --load-linux-policy --dict-wordlist /usr/share/dict/words
"""

from __future__ import annotations

import argparse
import json
import os
import re
import sys
from dataclasses import dataclass, asdict
from pathlib import Path
from typing import Iterable, Optional, Tuple, Dict, List


# ----------------------------
# Policy model
# ----------------------------

@dataclass
class PasswordPolicy:
    min_length: int = 14
    require_lower: bool = True
    require_upper: bool = True
    require_digit: bool = True
    require_special: bool = True

    # Maximum allowed identical chars in a row (e.g., "aaaa" => 4)
    max_consecutive_repeats: int = 3

    # Disallow password containing username (case-insensitive) if username length >= this
    forbid_username_substring: bool = True
    username_minlen_for_check: int = 3

    # Banned passwords (exact match, case-insensitive)
    banned_list_path: Optional[str] = None

    # Dictionary wordlist for substring checks (optional; can be large)
    dict_wordlist_path: Optional[str] = None
    dict_min_word_len: int = 4

    # Allow spaces? (passphrases)
    allow_spaces: bool = True


@dataclass
class CheckResult:
    password: str
    passed: bool
    failures: List[str]
    notes: List[str]


# ----------------------------
# Helpers
# ----------------------------

_SPECIAL_RE = re.compile(r"[^A-Za-z0-9\s]")
_LOWER_RE = re.compile(r"[a-z]")
_UPPER_RE = re.compile(r"[A-Z]")
_DIGIT_RE = re.compile(r"[0-9]")


def iter_lines(path: str) -> Iterable[str]:
    with open(path, "r", encoding="utf-8", errors="ignore") as f:
        for line in f:
            line = line.strip("\n\r")
            if not line:
                continue
            yield line


def load_banned_set(path: Optional[str]) -> set[str]:
    if not path:
        return set()
    p = Path(path)
    if not p.exists():
        raise FileNotFoundError(f"Banned list file not found: {path}")
    return {line.strip().lower() for line in iter_lines(path) if line.strip() and not line.lstrip().startswith("#")}


def load_dict_words(path: Optional[str], min_len: int) -> List[str]:
    if not path:
        return []
    p = Path(path)
    if not p.exists():
        raise FileNotFoundError(f"Dictionary wordlist not found: {path}")
    # Keep it memory-safe-ish: only store words meeting min length, and de-dup.
    words = set()
    for line in iter_lines(path):
        w = line.strip().lower()
        if len(w) >= min_len and w.isalpha():
            words.add(w)
    # Longer words first helps catch meaningful substrings earlier.
    return sorted(words, key=len, reverse=True)


def has_max_consecutive_repeats(pw: str, max_allowed: int) -> bool:
    if max_allowed <= 0:
        return False
    run = 1
    last = None
    for ch in pw:
        if ch == last:
            run += 1
            if run > max_allowed:
                return True
        else:
            last = ch
            run = 1
    return False


def contains_username(pw: str, username: Optional[str], minlen: int) -> bool:
    if not username:
        return False
    u = username.strip().lower()
    if len(u) < minlen:
        return False
    return u in pw.lower()


def dict_substring_hit(pw: str, dict_words: List[str]) -> Optional[str]:
    if not dict_words:
        return None
    pw_l = pw.lower()
    for w in dict_words:
        if w in pw_l:
            return w
    return None


# ----------------------------
# Linux local policy parsing (best-effort)
# ----------------------------

def parse_kv_file(path: Path) -> Dict[str, str]:
    """
    Parse simple key=value files (pwquality.conf style).
    Ignores comments and malformed lines.
    """
    out: Dict[str, str] = {}
    if not path.exists():
        return out
    for raw in iter_lines(str(path)):
        s = raw.strip()
        if not s or s.startswith("#"):
            continue
        # pwquality.conf can have "key = value" or "key=value"
        if "=" in s:
            k, v = s.split("=", 1)
            out[k.strip()] = v.strip()
    return out


def parse_login_defs(path: Path) -> Dict[str, str]:
    """
    Parse /etc/login.defs-ish: KEY   VALUE, with # comments.
    """
    out: Dict[str, str] = {}
    if not path.exists():
        return out
    for raw in iter_lines(str(path)):
        line = raw.strip()
        if not line or line.startswith("#"):
            continue
        # remove inline comment
        if "#" in line:
            line = line.split("#", 1)[0].strip()
        parts = line.split()
        if len(parts) >= 2:
            out[parts[0]] = parts[1]
    return out


def load_linux_policy_hints(base: PasswordPolicy) -> Tuple[PasswordPolicy, List[str]]:
    """
    Best-effort load of local Linux policy hints.
    Does NOT guarantee parity with PAM stack; it’s a practical approximation.
    """
    notes: List[str] = []
    policy = PasswordPolicy(**asdict(base))

    pwq = parse_kv_file(Path("/etc/security/pwquality.conf"))
    if pwq:
        notes.append("Loaded hints from /etc/security/pwquality.conf")
        # Common keys: minlen, dcredit, ucredit, lcredit, ocredit, maxrepeat
        if "minlen" in pwq:
            try:
                policy.min_length = max(policy.min_length, int(pwq["minlen"]))
            except ValueError:
                notes.append("Could not parse pwquality.conf minlen")

        if "maxrepeat" in pwq:
            try:
                policy.max_consecutive_repeats = int(pwq["maxrepeat"])
            except ValueError:
                notes.append("Could not parse pwquality.conf maxrepeat")

        # Credits in pwquality.conf: negative means REQUIRED; positive means "at most"
        # We'll interpret negative credit as requiring that class.
        def credit_requires(key: str) -> Optional[bool]:
            if key not in pwq:
                return None
            try:
                val = int(pwq[key])
                return val < 0
            except ValueError:
                return None

        req_lower = credit_requires("lcredit")
        req_upper = credit_requires("ucredit")
        req_digit = credit_requires("dcredit")
        req_other = credit_requires("ocredit")

        if req_lower is not None:
            policy.require_lower = req_lower
        if req_upper is not None:
            policy.require_upper = req_upper
        if req_digit is not None:
            policy.require_digit = req_digit
        if req_other is not None:
            policy.require_special = req_other

    login_defs = parse_login_defs(Path("/etc/login.defs"))
    if login_defs:
        # PASS_MIN_LEN is sometimes present; use as a floor
        if "PASS_MIN_LEN" in login_defs:
            try:
                policy.min_length = max(policy.min_length, int(login_defs["PASS_MIN_LEN"]))
                notes.append("Loaded hints from /etc/login.defs (PASS_MIN_LEN)")
            except ValueError:
                notes.append("Could not parse login.defs PASS_MIN_LEN")

    if not pwq and not login_defs:
        notes.append("No local Linux policy files found (pwquality.conf/login.defs). Using provided defaults.")

    return policy, notes


# ----------------------------
# Core check
# ----------------------------

def check_password(policy: PasswordPolicy, password: str, username: Optional[str], banned: set[str], dict_words: List[str]) -> CheckResult:
    failures: List[str] = []
    notes: List[str] = []

    pw = password.rstrip("\n\r")

    if not policy.allow_spaces and " " in pw:
        failures.append("Password contains spaces but policy disallows spaces.")

    if len(pw) < policy.min_length:
        failures.append(f"Password length {len(pw)} is less than minimum {policy.min_length}.")

    if policy.require_lower and not _LOWER_RE.search(pw):
        failures.append("Missing lowercase letter.")
    if policy.require_upper and not _UPPER_RE.search(pw):
        failures.append("Missing uppercase letter.")
    if policy.require_digit and not _DIGIT_RE.search(pw):
        failures.append("Missing digit.")
    if policy.require_special and not _SPECIAL_RE.search(pw):
        failures.append("Missing special character.")

    if has_max_consecutive_repeats(pw, policy.max_consecutive_repeats):
        failures.append(f"Contains more than {policy.max_consecutive_repeats} identical characters in a row.")

    if policy.forbid_username_substring and contains_username(pw, username, policy.username_minlen_for_check):
        failures.append("Contains username substring (case-insensitive).")

    if pw.lower() in banned:
        failures.append("Password is present in banned list (exact match, case-insensitive).")

    hit = dict_substring_hit(pw, dict_words)
    if hit:
        failures.append(f"Contains dictionary word substring: '{hit}'.")

    # NIST-aligned note (general): passphrases are encouraged; complexity rules are optional.
    if " " in pw and len(pw) >= policy.min_length:
        notes.append("Looks like a passphrase; long passphrases are generally preferable to short complex passwords.")

    return CheckResult(password=pw, passed=(len(failures) == 0), failures=failures, notes=notes)


# ----------------------------
# I/O
# ----------------------------

def load_users(users_path: Optional[str], single_username: Optional[str]) -> List[str]:
    if single_username:
        return [single_username]
    if not users_path:
        return [None]  # unknown user context
    p = Path(users_path)
    if not p.exists():
        raise FileNotFoundError(f"Users file not found: {users_path}")
    users = []
    for line in iter_lines(str(p)):
        s = line.strip()
        if s and not s.lstrip().startswith("#"):
            users.append(s)
    return users or [None]


def load_passwords(passwords_path: str) -> List[str]:
    p = Path(passwords_path)
    if not p.exists():
        raise FileNotFoundError(f"Passwords file not found: {passwords_path}")
    pwds = []
    for line in iter_lines(str(p)):
        s = line.strip()
        if s and not s.lstrip().startswith("#"):
            pwds.append(s)
    return pwds


def main() -> int:
    ap = argparse.ArgumentParser(description="Password Policy Tester (no authentication attempts)")
    ap.add_argument("--passwords", required=True, help="Path to passwords.txt (one password/passphrase per line)")
    ap.add_argument("--users", help="Optional users.txt (one username per line)")
    ap.add_argument("--username", help="Single username context (alternative to --users)")

    ap.add_argument("--report", help="Write JSON report to this path (default: stdout)")

    ap.add_argument("--min-length", type=int, default=14)
    ap.add_argument("--require-lower", action="store_true", default=False)
    ap.add_argument("--require-upper", action="store_true", default=False)
    ap.add_argument("--require-digit", action="store_true", default=False)
    ap.add_argument("--require-special", action="store_true", default=False)
    ap.add_argument("--no-spaces", action="store_true", default=False)
    ap.add_argument("--max-consecutive-repeats", type=int, default=3)
    ap.add_argument("--allow-username-substring", action="store_true", default=False)

    ap.add_argument("--banned-list", help="Optional banned passwords file (one entry per line)")
    ap.add_argument("--dict-wordlist", help="Optional dictionary wordlist for substring checks")
    ap.add_argument("--dict-min-word-len", type=int, default=4)

    ap.add_argument("--load-linux-policy", action="store_true", help="Best-effort load local Linux policy hints")

    args = ap.parse_args()

    base_policy = PasswordPolicy(
        min_length=args.min_length,
        # Default to NIST-friendly posture: don't force composition unless explicitly requested.
        require_lower=args.require_lower,
        require_upper=args.require_upper,
        require_digit=args.require_digit,
        require_special=args.require_special,
        max_consecutive_repeats=args.max_consecutive_repeats,
        forbid_username_substring=(not args.allow_username_substring),
        banned_list_path=args.banned_list,
        dict_wordlist_path=args.dict_wordlist,
        dict_min_word_len=args.dict_min_word_len,
        allow_spaces=(not args.no_spaces),
    )

    policy_notes: List[str] = []
    policy = base_policy
    if args.load_linux_policy:
        policy, policy_notes = load_linux_policy_hints(base_policy)

    banned = load_banned_set(policy.banned_list_path)
    dict_words = load_dict_words(policy.dict_wordlist_path, policy.dict_min_word_len)

    passwords = load_passwords(args.passwords)
    users = load_users(args.users, args.username)

    results: Dict[str, List[dict]] = {}
    for u in users:
        key = u if u is not None else "_no_user_context_"
        user_results = []
        for pw in passwords:
            r = check_password(policy, pw, u, banned, dict_words)
            user_results.append(asdict(r))
        results[key] = user_results

    report_obj = {
        "policy": asdict(policy),
        "policy_notes": policy_notes,
        "counts": {
            "passwords_tested": len(passwords),
            "user_contexts": len(users),
            "banned_list_loaded": len(banned),
            "dict_words_loaded": len(dict_words),
        },
        "results": results,
    }

    out = json.dumps(report_obj, indent=2)
    if args.report:
        Path(args.report).write_text(out, encoding="utf-8")
        print(f"Wrote report to: {args.report}")
    else:
        print(out)

    # exit nonzero if any failures exist
    any_failed = any(
        not entry["passed"]
        for user_key in results
        for entry in results[user_key]
    )
    return 2 if any_failed else 0


if __name__ == "__main__":
    raise SystemExit(main())
